    <script src="https://unpkg.com/leaflet@1.9.4/dist/leaflet.js"></script>
    <!-- Firebase SDK -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
        import { getFirestore, collection, addDoc, getDocs, deleteDoc, doc, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";
        import { getAuth, signInWithEmailAndPassword, signOut, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-auth.js";

        // 1. BURAYA FIREBASE'DEN ALDIĞIN CONFIG KODUNU YAPIŞTIR
        const firebaseConfig = {
            apiKey: "AIzaSyDB5gBNWtzrZAk4do8HdC0sR3_rB37Iqhs",
            authDomain: "kampus-gorus-haritasi77.firebaseapp.com",
            projectId: "kampus-gorus-haritasi77",
            storageBucket: "kampus-gorus-haritasi77.firebasestorage.app",
            messagingSenderId: "761474558310",
            appId: "1:761474558310:web:8d51681b37c8d1cd27c0a7"
        };

        const app = initializeApp(firebaseConfig);
        const db = getFirestore(app);
        const auth = getAuth(app);

        // 2. HARİTA AYARLARI (JPG Boyutunu Buraya Gir - Örn: 2000x1500)
        const imgWidth = 2500;  // JPG resminin genişliği (piksel)
        const imgHeight = 1768; // JPG resminin yüksekliği (piksel)

        const map = L.map('map', {
            crs: L.CRS.Simple,
            minZoom: -2,
            maxZoom: 2
        });

        const bounds = [[0, 0], [imgHeight, imgWidth]];
        L.imageOverlay('harita.jpg', bounds).addTo(map);
        map.fitBounds(bounds);

        let tempMarker = null;
        let selectedCoords = null;
        let adminMarkers = [];

        // Öğrenci Haritaya Tıkladığında
        map.on('click', function(e) {
            if (auth.currentUser) return; // Admin modundaysa yeni pin ekleme formu açma
            selectedCoords = e.latlng;
            if (tempMarker) map.removeLayer(tempMarker);
            tempMarker = L.marker(selectedCoords).addTo(map);
            document.getElementById('feedbackModal').style.display = 'flex';
        });

        // Form Gönderme
        window.submitFeedback = async function() {
            const type = document.getElementById('type').value;
            const comment = document.getElementById('comment').value.trim();
            if (!comment) return alert("Lütfen bir mesaj yazın.");

            try {
                await addDoc(collection(db, "pins"), {
                    type: type,
                    comment: comment,
                    lat: selectedCoords.lat,
                    lng: selectedCoords.lng,
                    createdAt: serverTimestamp()
                });
                alert("Geri bildiriminiz başarıyla iletildi. Teşekkür ederiz!");
                closeModal();
            } catch (err) {
                alert("Hata oluştu: " + err.message);
            }
        };

        window.closeModal = function() {
            document.getElementById('feedbackModal').style.display = 'none';
            document.getElementById('comment').value = '';
            if (tempMarker) map.removeLayer(tempMarker);
        };

        // Admin Giriş / Çıkış İşlemleri
        window.openLogin = () => document.getElementById('loginModal').style.display = 'flex';
        window.closeLogin = () => document.getElementById('loginModal').style.display = 'none';

        window.adminLogin = async function() {
            const email = document.getElementById('adminEmail').value;
            const pass = document.getElementById('adminPass').value;
            try {
                await signInWithEmailAndPassword(auth, email, pass);
                closeLogin();
            } catch (err) {
                alert("Giriş başarısız: " + err.message);
            }
        };

        window.adminLogout = () => signOut(auth);

        // Admin Durumunu Dinle ve Pinleri Yükle
        onAuthStateChanged(auth, async (user) => {
            if (user) {
                document.getElementById('loginBtn').style.display = 'none';
                document.getElementById('adminControls').style.display = 'flex';
                loadPins();
            } else {
                document.getElementById('loginBtn').style.display = 'block';
                document.getElementById('adminControls').style.display = 'none';
                adminMarkers.forEach(m => map.removeLayer(m));
                adminMarkers = [];
            }
        });

        // Veritabanındaki Pinleri Haritaya Çiz (Sadece Admin Görür)
        async function loadPins() {
            adminMarkers.forEach(m => map.removeLayer(m));
            adminMarkers = [];
            const querySnapshot = await getDocs(collection(db, "pins"));
            
            querySnapshot.forEach((docSnap) => {
                const data = docSnap.data();
                const color = data.type === 'Talep' ? 'red' : 'green';
                
                const markerIcon = L.divIcon({
                    className: 'custom-pin',
                    html: `<div style="background:${color}; width:18px; height:18px; border-radius:50%; border:3px solid white; box-shadow:0 0 5px rgba(0,0,0,0.5);"></div>`
                });

                const marker = L.marker([data.lat, data.lng], { icon: markerIcon }).addTo(map);
                marker.bindPopup(`
                    <b>Tür:</b> ${data.type}<br>
                    <b>Mesaj:</b> ${data.comment}<br><br>
                    <button onclick="deletePin('${docSnap.id}')" style="background:#dc3545; color:white; border:none; padding:4px 8px; border-radius:4px; cursor:pointer;">Sil</button>
                `);
                adminMarkers.push(marker);
            });
        }

        window.deletePin = async function(id) {
            if(confirm("Bu pini silmek istediğinize emin misiniz?")) {
                await deleteDoc(doc(db, "pins", id));
                loadPins();
            }
        };
